"""H5b timing micro-measurement (INFORMATION ONLY - not the row-439 campaign, no workload/contention/scheduler cell):
python timing.py <research/s1 code dir> <manifest.jsonl> <label>. For each golden fixture and each pass owner of the
chain, the number of c.load_pinned("e1_sim_pinned") calls in ONE call (= one pass) and its wall time (min of 5)."""
import json
import sys
import time
from pathlib import Path

CODE, MANIFEST, LABEL = Path(sys.argv[1]).resolve(), Path(sys.argv[2]), sys.argv[3]
sys.path.insert(0, str(CODE))

import s1_common as c  # noqa: E402
import s1_journal as j  # noqa: E402
import s1_seal as z  # noqa: E402
import s1_trial as t  # noqa: E402

loads = []
real_lp = c.load_pinned


def counted(name, *a, **kw):
    loads.append(name)
    return real_lp(name, *a, **kw)


c.load_pinned = counted
lp_time = [0.0]


def main():
    print(f"# {LABEL}: code {CODE}")
    print(f"{'fixture':26} {'pass owner':18} {'load_pinned/pass':>16} {'min wall ms (5)':>16}")
    for line in MANIFEST.read_text(encoding="utf-8").splitlines():
        m = json.loads(line)
        if m["name"] not in ("lots-stopped", "six-month-timeout", "timeout-99", "nvs"):
            continue
        root = Path(m["root"])
        mb = root / "mailbox.sqlite3"
        c.MAILBOX = mb
        trs = t.trials(t.scan(mb))
        (tid,) = [x for x, tr_ in trs.items() if tr_.closed is None]
        rows = j.read_trial(tid, trs[tid].genesis, mb)
        out = root / "s1_out"
        pr = json.loads((out / "prereg.json").read_text(encoding="utf-8"))
        start, cal, bundle = c.aware(pr["start"]), pr["calendar"], trs[tid].bundle
        stop = j.first(rows, "TRIAL-STOP")
        pre = [x for x in rows if x.seq < stop.seq]
        owners = {"stop_record": lambda: z.stop_record(out, pre, start, cal, bundle),
                  "sampled_seals": lambda: z.sampled_seals(out, rows, cal, start, upto_seq=stop.seq),
                  "nvs": lambda: z.nvs(out, rows, start, cal, bundle),
                  "stop_ready": lambda: z.stop_ready(out, pre, start, cal, bundle, c.aware(m["now"])),
                  "eligible_sessions": lambda: z.eligible_sessions(start, cal)}
        for name, fn in owners.items():
            n0 = len(loads)
            fn()
            n = sum(1 for x in loads[n0:] if x == "e1_sim_pinned")
            best = min(_timed(fn) for _ in range(5))
            print(f"{m['name']:26} {name:18} {n:16d} {best * 1000:16.1f}")


def _timed(fn):
    t0 = time.perf_counter()
    fn()
    return time.perf_counter() - t0


if __name__ == "__main__":
    main()
