def sha(p: Path) -> str:
    return hashlib.sha256(p.read_bytes()).hexdigest()


def copy(tag) -> Path:
    root = SCR / tag
    if root.exists():
        shutil.rmtree(root)
    shutil.copytree(SRC, root / "research" / "s1", ignore=shutil.ignore_patterns("__pycache__", ".pytest_cache"))
    return root


def apply(root, fname, reps):
    p = root / "research" / "s1" / fname
    raw = p.read_bytes()
    crlf = b"\r\n" in raw
    s = raw.decode("utf-8").replace("\r\n", "\n")
    for a, b in reps:
        assert s.count(a) == 1, (fname, a[:60], s.count(a))
        s = s.replace(a, b)
    p.write_bytes((s.replace("\n", "\r\n") if crlf else s).encode("utf-8"))
    return sha(p)
