"""H5b golden fixtures (scratch only, never in the worktree): each case builds a connected s1_out/ under its own
tmp_path through the REAL product gates (conftest.s1_state), applies its damage, and appends {name, root, now} to the
manifest named by H5B_GOLD_MANIFEST. Run once, against the BASE copy; the fixtures are then evaluated read-only by
golden_eval.py under the BASE code and under the H5b code (same fixture bytes)."""
import json
import os
from pathlib import Path

import pytest
import s1_journal as j
import s1_seal as z
import s1_slot as sl
import s1_trial as t
from conftest import s1_jappend, s1_state

START = "2026-10-26T13:30:00+00:00"
LOTS99 = [{"symbol": f"S{i % 5}", "qty": 10, "px": 100.0, "buy_day": 0, "sell": (1, 80.0) if i < 99 else None}
          for i in range(100)]
LOTS40 = [{"symbol": f"S{i % 5}", "qty": 10, "px": 100.0, "buy_day": 0, "sell": (1, 80.0)} for i in range(40)]


def mb(root):
    return Path(root) / "mailbox.sqlite3"


def first_seal_files(root):
    out = Path(root) / "s1_out"
    tr = t.open_trial(mb(root))
    rows = j.read_trial(tr.trial_id, tr.genesis, mb(root))
    pr = json.loads((out / "prereg.json").read_text(encoding="utf-8"))
    import s1_common as c
    day, (q, _b) = sorted(z.sampled_seals(out, rows, pr["calendar"], c.aware(pr["start"])).items())[0]
    return sl.names(out, sl.slot_of(q.subject), day)


def build(name, root):
    if name == "lots-stopped":
        return s1_state(root, START, score_ready=True)
    if name == "six-month-timeout":
        return s1_state(root, START, score_ready=True, lots=LOTS40)
    if name == "timeout-99":
        return s1_state(root, START, score_ready=True, lots=LOTS99, seal_days=[0, 1, 2])
    if name == "no-stop-three-qualified":
        return s1_state(root, START, score_ready=True, stop=False, lots=LOTS99, seal_days=[0, 1, 2])
    if name == "no-seal-stage":
        return s1_state(root, START, score_ready=True, stop=False, seal_stage=False, lots=LOTS99,
                        seal_days=[0, 1, 2])
    if name == "nvs":
        from test_s1_a3_nvs import nvs_chain
        return nvs_chain(root)
    if name == "pending":
        st_ = s1_state(root, START, score_ready=True, stop=False, lots=LOTS99, seal_days=[1, 2])
        import s1_common as c
        d0 = z.eligible_sessions(c.aware(START), st_["calendar"])[0]
        s = z.schedule(d0, st_["calendar"])[0]
        s1_jappend(root, "SLOT-INTENT", j.slot_subject(s), at=s, op="1" * 32, actor="2" * 32)
        return st_
    if name.startswith("damaged-"):
        st_ = s1_state(root, START, score_ready=True)
        n = first_seal_files(root)
        if name == "damaged-resolution-appended":
            n["resolution"].write_bytes(n["resolution"].read_bytes() + b" ")
        elif name == "damaged-backup-same-size":
            p = n["capture"] / "backup.sqlite3"
            st = p.stat()
            b = bytearray(p.read_bytes())
            b[-1] ^= 0x01
            p.write_bytes(bytes(b))
            os.utime(p, ns=(st.st_atime_ns, st.st_mtime_ns))
        elif name == "damaged-backup-missing":
            (n["capture"] / "backup.sqlite3").unlink()
        elif name == "damaged-resolution-malformed":
            n["resolution"].write_bytes(b"{not json")
        elif name == "damaged-projection-changed":
            p = n["capture"] / "projection.json"
            p.write_bytes(p.read_bytes() + b" ")
        return st_
    raise AssertionError(name)


CASES = ["lots-stopped", "six-month-timeout", "timeout-99", "no-stop-three-qualified", "no-seal-stage", "nvs",
         "pending", "damaged-resolution-appended", "damaged-backup-same-size", "damaged-backup-missing",
         "damaged-resolution-malformed", "damaged-projection-changed"]


@pytest.mark.parametrize("name", CASES)
def test_build(tmp_path, name):
    st_ = build(name, tmp_path)
    with open(os.environ["H5B_GOLD_MANIFEST"], "a", encoding="utf-8") as f:
        f.write(json.dumps({"name": name, "root": str(tmp_path), "now": st_["now"].isoformat()}) + "\n")

def row(x):
    return None if x is None else [x.seq, x.sha, x.subject, x.kind]


def safe(fn):
    try:
        return {"ok": enc(fn())}
    except BaseException as e:  # noqa: BLE001 - every outcome is the golden, exceptions included
        if isinstance(e, KeyboardInterrupt):
            raise
        ident = getattr(e, "identity", None)
        return {"exc": type(e).__name__, "msg": str(e), "identity": list(ident) if ident else None}


def enc(v):
    if isinstance(v, bytes):
        return {"b64": base64.b64encode(v).decode("ascii")}
    if isinstance(v, dict):
        return {str(k): enc(x) for k, x in v.items()}
    if isinstance(v, (list, tuple)):
        return [enc(x) for x in v]
    if hasattr(v, "seq") and hasattr(v, "sha"):
        return {"row": row(v)}
    if hasattr(v, "isoformat"):
        return v.isoformat()
    return v


def seals_enc(s):
    return {d: {"q": row(q), "bytes_b64": base64.b64encode(b).decode("ascii")} for d, (q, b) in sorted(s.items())}


def chain(out, rows, start, cal, bundle, nows):
    stop = j.first(rows, "TRIAL-STOP")
    res = {"stop_record": safe(lambda: z.stop_record(out, rows, start, cal, bundle)),
           "nvs": safe(lambda: z.nvs(out, rows, start, cal, bundle)),
           "sampled_seals": safe(lambda: seals_enc(z.sampled_seals(out, rows, cal, start))),
           "stop_ready": {n.isoformat(): safe(lambda n=n: z.stop_ready(out, rows, start, cal, bundle, n))
                          for n in nows}}
    if stop is not None:
        res["sampled_seals_upto_stop"] = safe(lambda: seals_enc(z.sampled_seals(out, rows, cal, start,
                                                                                upto_seq=stop.seq)))
        res["sampled_history"] = safe(lambda: z.sampled_history(
            rows, stop, z.sampled_seals(out, rows, cal, start, upto_seq=stop.seq)))
    return res


def main():
    got = {}
    for line in MANIFEST.read_text(encoding="utf-8").splitlines():

        mb = root / "mailbox.sqlite3"

        out = root / "s1_out"
        trs = t.trials(t.scan(mb))
        (tid,) = [x for x, tr_ in trs.items() if tr_.closed is None]
        tr = trs[tid]
        rows = j.read_trial(tid, tr.genesis, mb)
        pr = json.loads((out / "prereg.json").read_text(encoding="utf-8"))
        start, cal, bundle = c.aware(pr["start"]), pr["calendar"], tr.bundle
        now = c.aware(m["now"])

        if stop is not None:
            case["pre_stop"] = chain(out, [x for x in rows if x.seq < stop.seq], start, cal, bundle, nows)

"""Summarize a golden JSON: per fixture, the outcome kind of each chain result (ok value shape or exception type)."""
import json
import sys
from pathlib import Path

g = json.loads(Path(sys.argv[1]).read_text(encoding="ascii"))


def kind(r):
    if "exc" in r:
        return f"EXC {r['exc']}: {r['msg'][:70]}"
    v = r["ok"]
    if isinstance(v, dict):
        if "status" in v:
            return f"record {v['status']}/{v['rule']} stop_close={v['stop_close']}"
        return f"dict[{len(v)}]"
    return repr(v)[:70]


for name, case in g.items():
    print(f"== {name}: {len(case['rows'])} rows, eligible {kind(case['eligible_sessions'])}, "
          f"{len(case['prefixes'])} prefixes")
    for part in ("full", "pre_stop"):
        if part not in case:
            continue
        for k, r in case[part].items():
            if k == "stop_ready":
                print(f"   {part}.stop_ready: " + " | ".join(kind(x) for x in r.values()))
            else:
                print(f"   {part}.{k}: {kind(r)}")
    pk = {}
    for s, p in case["prefixes"].items():
        for k, r in p.items():
            pk.setdefault(k, []).append(kind(r).split(":")[0][:30])
    for k, v in pk.items():
        print(f"   prefixes.{k}: {sorted(set(v))}")
