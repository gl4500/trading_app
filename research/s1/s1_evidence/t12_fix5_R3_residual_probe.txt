(R-1 no longer blocks it: the finalizer's MISSED records of the never-claimed opening slots are consistent).
Variants:
  V1 probe7 as written: sealed at 20:45, NO later opening slot ran; delete the seal, both opening records, every
     resolution and report of the opening close, opening_receipt.json (claims + capture dir kept).
  V2 a later opening slot (20:47) ran after the seal (R-3 after-seal claim/resolution/report), then the V1 deletion
     set (which also removes the after-seal resolution and report; its claim stays).
  V3 as V2 but the after-seal report and resolution are kept (only the sealing slot's records deleted).
Each prints whether a second, different opening seal is written at the 00:00 UTC opening slot, and any abort."""
import shutil
import sys
from datetime import timedelta

import pytest

sys.path.insert(0, "C:/Users/gl450/trading_app-s1/research/s1/s1_tests")
import s1_checkpoint as ck, s1_common as c, s1_seal as z
from realconf import S1_RECEIPT, S1_PROCESS, Ledger, _Clock, s1_plan_audit, s1_opening_checkpoint

START = "2026-10-26T13:30:00+00:00"
FMT = "%Y%m%dT%H%M%SZ"


def _run(tmp_path, lg, day, cal, i):
    sch = z.schedule(day, cal)[i]
    act = sch + timedelta(seconds=30)
    with _Clock(act):
        return ck.run_opening_checkpoint(tmp_path, lg.path, sch, lambda: S1_RECEIPT, lambda: dict(S1_PROCESS),
                                         tmp_path / "crash.log", now=lambda: act)


@pytest.mark.parametrize("variant", ["V1", "V2", "V3"])
def test_crash_like_residual(tmp_path, variant):
    start, cal, oc, _ = s1_plan_audit(tmp_path, START)
    lg = Ledger(tmp_path / "l.sqlite3")
    day = next(d for d in cal if d["date"] == oc)
    s1_opening_checkpoint(tmp_path, lg, oc, cal, S1_RECEIPT)
    out = tmp_path / "s1_out"
    first = c.sha256_file(out / "seals" / f"{oc}.json")
    if variant in ("V2", "V3"):
        try:
            _run(tmp_path, lg, day, cal, 1)
        except SystemExit as e:
            print(variant, "after-seal slot:", str(e)[:120])
    (out / "seals" / f"{oc}.json").unlink()
    for d in ("registry", "witnesses"):
        shutil.rmtree(out / d)
    sealing = z.schedule(day, cal)[0].strftime(FMT)
    for p in (out / "captures" / oc).glob("*"):
        if p.name.endswith((".resolution.json", ".report.json")) and (variant != "V3" or p.name.startswith(sealing)):
            p.unlink()
    (out / "opening_receipt.json").unlink()
    print(variant, "left:", sorted(p.relative_to(out).as_posix() for p in out.rglob("*") if p.is_file()
                                   and not p.relative_to(out).as_posix().startswith(("audit/", "captures/" + oc + "/" + sealing + "/"))))
    lg.trade("Z", "BUY", 1, 10.0, (z.schedule(day, cal)[8] - timedelta(minutes=1)).isoformat())
    lg.set_portfolio("Z", 1)
    try:
        rep = _run(tmp_path, lg, day, cal, 8)
        print(variant, "RESULT seal", rep["seal"], "| second, different seal:",
              rep["seal"].get("written") is True and rep["seal"].get("sha256") != first)
    except SystemExit as e:
        print(variant, "RESULT refused:", str(e)[:200])
    print(variant, "ABORT", (c.load_abort(out) or {}).get("reason"))
